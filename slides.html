function addMessage(role, text) {
  const div = document.createElement("div");
  div.className = `message ${role}`;
  div.textContent = text;
  chatMessages.appendChild(div);
  chatMessages.scrollTop = chatMessages.scrollHeight;
  return div;
}

How hands work: a loop

You aska question
→
Model"I need to look this up"
→
[TOOL weather "Tokyo"]it writes a marker
→
Your coderuns fetch()
→
Modelanswers from the result

The model never touches the network. It asks. Your JavaScript does.

The whole trick: a regex and a loop

// Model outputs: [TOOL wikipedia "Eiffel Tower"]
function parseToolCall(text) {
  const match = text.match(/\[TOOL\s+(\w+)(?:\s+"([^"]*)")?\]/i);
  if (!match) return null;
  return { tool: match[1].toLowerCase(), arg: match[2] || "" };
}

for (let round = 0; round < MAX_TOOL_ROUNDS; round++) {
  const { reply } = await getModelResponse();
  const toolCall = parseToolCall(reply);
  if (!toolCall) break;                       // no tool: that was the answer

  const result = await executeTool(toolCall.tool, toolCall.arg);
  chatHistory.push({ role: "user",
    content: `[TOOL RESULT]\n${result.text}\n[/TOOL RESULT]` });
}
